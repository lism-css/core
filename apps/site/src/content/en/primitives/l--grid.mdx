### Switching grid areas at breakpoints

<Preview>
  <PreviewTitle>Switching grid areas at breakpoints</PreviewTitle>
  <PreviewArea resize p="15">
    <Grid g="15" gtc={['1fr 1fr', '8em 1fr 8em']} gta={[`'main main' 'left right'`, `'left main right'`]}>
      <Box ga="left" p="15" bgc="base-2">Left</Box>
      <Box ga="main" p="15" bd>Center</Box>
      <Box ga="right" p="15" bgc="base-2">Right</Box>
    </Grid>
  </PreviewArea>
  <PreviewCode slot="tab" label="JSX">
    ```jsx
    <Grid g="15" gtc={['1fr 1fr', '8em 1fr 8em']} gta={[`'main main' 'left right'`, `'left main right'`]}>
      <Box ga="left" p="15" bgc="base-2">
        Left
      </Box>
      <Box ga="main" p="15" bd>
        Center
      </Box>
      <Box ga="right" p="15" bgc="base-2">
        Right
      </Box>
    </Grid>
    ```
  </PreviewCode>
  <PreviewCode slot="tab" label="HTML">
    ```html
    <div
      class="l--grid -gta -gta_sm -gtc -gtc_sm -g:15"
      style="--gta: 'main main' 'left right'; --gta_sm: 'left main right'; --gtc: 1fr 1fr; --gtc_sm: 8em 1fr 8em"
    >
      <div class="l--box -p:15 -bgc:base-2 -ga" style="--ga: left">Left</div>
      <div class="l--box -p:15 -bd -ga" style="--ga: main">Center</div>
      <div class="l--box -p:15 -bgc:base-2 -ga" style="--ga: right">Right</div>
    </div>
    ```
  </PreviewCode>
</Preview>


### Overlapping child elements

This example stacks elements using `grid-area:1/1` (`-ga:1/1`) without `position:absolute`.

<Preview>
  <PreviewTitle>Overlapping Grid child elements</PreviewTitle>
  <PreviewArea p="15">
    <Grid ar="16/9">
      <Frame ga="1/1" w="100%" h="100%" pos="relative">
        <Media src="https://cdn.lism-css.com/img/KRQPW50jXaLu.webp" alt="" inferSize />
        <Layer style={{ backdropFilter: 'blur(10px)' }} bgc="rgb(0 0 0 / 40%)" />
      </Frame>
      <Center ga="1/1" fz="3xl" z="1" c="white">SAMPLE TEXT</Center>
    </Grid>
  </PreviewArea>
  <PreviewCode slot="tab" label="JSX">
    ```jsx 'ga="1/1"'
    <Grid ar="16/9">
      <Frame ga="1/1" w="100%" h="100%" pos="relative">
        <Media src="https://cdn.lism-css.com/img/KRQPW50jXaLu.webp" alt="" inferSize />
        <Layer style={{ backdropFilter: 'blur(10px)' }} bgc="rgb(0 0 0 / 40%)" />
      </Frame>
      <Center ga="1/1" fz="3xl" z="1" c="white">
        SAMPLE TEXT
      </Center>
    </Grid>
    ```
  </PreviewCode>
  <PreviewCode slot="tab" label="HTML">
    ```html "-ga:1/1"
    <div class="l--grid -ar:16/9">
      <div class="l--frame -ga:1/1 -w:100% -h:100% -pos:relative">
        <img src="https://cdn.lism-css.com/img/KRQPW50jXaLu.webp" />
        <div class="is--layer -bgc" style="backdrop-filter:blur(10px);--bgc:rgb(0 0 0 / 40%)"></div>
      </div>
      <div class="l--center -ga:1/1 -fz:3xl -z:1 -c" style="--c:var(--white)">
        SAMPLE TEXT
      </div>
    </div>
    ```
  </PreviewCode>
</Preview>
