export const ReelImages = ({ length=6, hasID, isV, ...props }) => {
  const imageUrls = [
    'https://images.unsplash.com/photo-1615209853186-e4bd66602508?auto=format&fit=max&w=1920&q=80',
    'https://images.unsplash.com/photo-1449034446853-66c86144b0ad?auto=format&fit=max&w=1920&q=80',
    'https://images.unsplash.com/photo-1534868489953-ef37c20abb25?auto=format&fit=max&w=1920&q=80',
    'https://images.unsplash.com/photo-1761597327650-fa885c49bef4?auto=format&fit=max&w=1920&q=80',
    'https://images.unsplash.com/photo-1578637387939-43c525550085?auto=format&fit=max&w=1920&q=80',
    'https://images.unsplash.com/photo-1469854523086-cc02fe5d8800?auto=format&fit=max&w=1920&q=80',
  ];
  return (
    <>
      {Array.from({length}, (v, i) => i).map((i) => {
        const src = imageUrls[i];
        return <Frame id={hasID ? `image-${i+1}` : ''} ar="16/9" key={i} {...props}>
          <img src={src} alt="" loading="lazy" width="960" height="640" />
        </Frame>
      })}
    </>
  );
}

## Disabling the reel at larger screen sizes

To show the reel only on small screens, set `gaf` (`grid-auto-flow`) back to `row`.
The following example demonstrates switching to a 2-column layout by setting `gtc` (`grid-template-columns`) to `repeat(2, 1fr)` at the same breakpoint.

<Preview>
  <PreviewTitle>Example: disable reel only at `~@md`<SwipeHelp /></PreviewTitle>
  <PreviewArea resize p="10">
    <Grid hasSnap ov="auto" g="15" gaf={['column', null, 'row']} gac="75%" gtc={[null, null, 'repeat(2, 1fr)']}>
      <ReelImages />
    </Grid>
  </PreviewArea>
  <PreviewCode>
    ```jsx "gaf" "gtc"
    <Grid hasSnap ov="auto" g="15" gaf={['column', null, 'row']} gac="75%" gtc={[null, null, 'repeat(2, 1fr)']}>
    ```
